### F05 (fanout): fail -> pass

> For each year, show the number of delivered orders, the total item revenue and the total amount paid.

v1 (repeat 1, 0/3 passed):
```sql
SELECT 
    strftime('%Y', o.order_purchase_timestamp) AS year,
    COUNT(DISTINCT o.order_id) AS delivered_orders,
    SUM(oi.price) AS total_item_revenue,
    SUM(op.payment_value) AS total_amount_paid
FROM orders o
JOIN order_items oi   ON o.order_id = oi.order_id
JOIN order_payments op ON o.order_id = op.order_id
WHERE o.order_delivered_customer_date IS NOT NULL
GROUP BY year
ORDER BY year;
```
v2 (repeat 3, 2/3 passed):
```sql
WITH delivered_orders AS (
    SELECT
        order_id,
        strftime('%Y', order_purchase_timestamp) AS year
    FROM orders
    WHERE order_status = 'delivered'
),
order_item_rev AS (
    SELECT
        order_id,
        SUM(price) AS item_revenue
    FROM order_items
    GROUP BY order_id
),
order_payment_sum AS (
    SELECT
        order_id,
        SUM(payment_value) AS amount_paid
    FROM order_payments
    GROUP BY order_id
)
SELECT
    d.year,
    COUNT(d.order_id) AS delivered_orders,
    SUM(COALESCE(oir.item_revenue, 0)) AS total_item_revenue,
    SUM(COALESCE(ops.amount_paid, 0)) AS total_amount_paid
FROM delivered_orders d
LEFT JOIN order_item_rev oir ON d.order_id = oir.order_id
LEFT JOIN order_payment_sum ops ON d.order_id = ops.order_id
GROUP BY d.year
ORDER BY d.year;
```
gold:
```sql
WITH i AS (SELECT order_id, SUM(price) AS rev FROM order_items GROUP BY order_id),
p AS (SELECT order_id, SUM(payment_value) AS paid FROM order_payments GROUP BY order_id)
SELECT strftime('%Y', o.order_purchase_timestamp) AS year, COUNT(*) AS orders,
       ROUND(SUM(i.rev), 2) AS revenue, ROUND(SUM(p.paid), 2) AS paid
FROM orders o LEFT JOIN i ON o.order_id = i.order_id LEFT JOIN p ON o.order_id = p.order_id
WHERE o.order_status = 'delivered' GROUP BY year
```
