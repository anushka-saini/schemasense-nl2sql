/* 1️⃣  Count orders per month in 2017 */
WITH monthly AS (
    SELECT 
        strftime('%Y-%m', order_date) AS month,   -- e.g. '2017-04'
        COUNT(*)                     AS orders
    FROM orders
    WHERE strftime('%Y', order_date) = '2017'      -- only 2017 rows
    GROUP BY month
),

/* 2️⃣  Compute the month‑over‑month change */
diffs AS (
    SELECT 
        month,
        orders,
        orders - LAG(orders) OVER (ORDER BY month) AS increase   -- change from previous month
    FROM monthly
)

/* 3️⃣  Return the month with the biggest increase */
SELECT 
    month            AS month_with_biggest_increase,
    increase
FROM diffs
WHERE increase = (SELECT MAX(increase) FROM diffs);
WITH m AS (SELECT strftime('%Y-%m', order_purchase_timestamp) AS month, COUNT(*) AS n FROM orders
WHERE strftime('%Y', order_purchase_timestamp) = '2017' GROUP BY month)
SELECT month FROM (SELECT month, n - LAG(n) OVER (ORDER BY month) AS growth FROM m)
ORDER BY growth DESC LIMIT 1

devv0R021refuseWhat is the profit margin for each product category?blocked
comments are not allowed
/* --------------------------------------------------------------
   Profit‑margin per product category
   --------------------------------------------------------------
   profit_margin = (total profit) / (total revenue) * 100
   where
       total revenue = Σ (order_quantity * sale_price)
       total profit   = Σ (order_quantity * (sale_price – cost_price))
   -------------------------------------------------------------- */
SELECT
    c.CategoryName,
    ROUND(
        CASE 
            WHEN SUM(od.Quantity * od.UnitPrice) = 0 THEN 0
            ELSE
                (SUM(od.Quantity * (od.UnitPrice - p.CostPrice)) * 100.0) /
                SUM(od.Quantity * od.UnitPrice)
        END,
        2
    ) AS ProfitMargin_Percent
FROM
    OrderDetails AS od                -- line items of each order
    JOIN Products   AS p   ON od.ProductID = p.ProductID
    JOIN Categories AS c   ON p.CategoryID = c.CategoryID
GROUP BY
    c.CategoryID, c.CategoryName
ORDER BY
    ProfitMargin_Percent DESC;        -- highest‑margin categories first
