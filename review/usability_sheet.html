WITH delivered_orders AS (
    SELECT order_id
    FROM orders
    WHERE order_status = 'delivered'
),
seller_rev AS (
    SELECT oi.seller_id,
           SUM(oi.price) AS revenue
    FROM order_items oi
    JOIN delivered_orders d ON oi.order_id = d.order_id
    GROUP BY oi.seller_id
),
total_rev AS (
    SELECT SUM(revenue) AS total
    FROM seller_rev
),
top10_rev AS (
    SELECT SUM(revenue) AS top10
    FROM (
        SELECT revenue
        FROM seller_rev
        ORDER BY revenue DESC
        LIMIT 10
    )
)
SELECT ROUND((top10_rev.top10 / total_rev.total) * 100, 2) AS top10_percentage
FROM top10_rev, total_rev;
